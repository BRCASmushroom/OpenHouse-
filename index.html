function doGet(e) { var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet(); var action = e.parameter.action; // 初始化或取得目前狀態 if (action === 'get') { var data = sheet.getDataRange().getValues(); var totalRegistered = 0; var notifiedCount = 0; var admittedNumber = 0; if (data.length > 1) { totalRegistered = data.length - 1; // 計算已通知的人數 for (var i = 1; i < data.length; i++) { if (data[i][1] === 'notified' || data[i][1] === 'admitted') { notifiedCount = Math.max(notifiedCount, data[i][0]); } } admittedNumber = notifiedCount; } var output = { totalRegistered: totalRegistered, notifiedCount: notifiedCount, admittedNumber: admittedNumber }; return ContentService.createTextOutput(JSON.stringify(output)) .setMimeType(ContentService.MimeType.JSON); } // 民眾登記候位 if (action === 'register') { var lastRow = sheet.getLastRow(); var newId = lastRow === 0 ? 1 : lastRow; // 如果只有標題行，第一筆是 1 sheet.appendRow([newId, 'registered', new Date()]); return ContentService.createTextOutput(JSON.stringify({ success: true, myNumber: newId })) .setMimeType(ContentService.MimeType.JSON); } // 工作人員發送遞補通知 if (action === 'notify') { var slots = parseInt(e.parameter.slots || 1); var data = sheet.getDataRange().getValues(); var currentNotified = 0; // 找出目前已通知到幾號 for (var i = 1; i < data.length; i++) { if (data[i][1] === 'notified') { currentNotified = Math.max(currentNotified, data[i][0]); } } var targetLimit = currentNotified + slots; var totalRows = data.length - 1; if (targetLimit > totalRows) targetLimit = totalRows; // 更新指定範圍的狀態為 notified for (var i = 1; i < data.length; i++) { if (data[i][0] <= targetLimit && data[i][1] === 'registered') { sheet.getRange(i + 1, 2).setValue('notified'); } } return ContentService.createTextOutput(JSON.stringify({ success: true, notifiedCount: targetLimit })) .setMimeType(ContentService.MimeType.JSON); } }
